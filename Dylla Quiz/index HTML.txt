                    let total = 0;
                    masterQuestions.forEach(q => {
                        const selectedOptIdx = userAnswers[q.id];
                        if(selectedOptIdx !== undefined) {

                    });

        // Sistem Navigasi Halaman Halaman
        function navigateTo(screenId) {
            document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
            document.getElementById(screenId).classList.add('active');
            // Auto scroll top tiap pindah halaman
            document.getElementById(screenId).scrollTop = 0;
        }

        function startQuiz() {
            navigateTo('screen-3');
        }

        // Render Komponen Kuis secara Dinamis & Aman
        function renderQuiz() {
            const containerP1 = document.getElementById('questions-container-p1');
            const containerP2 = document.getElementById('questions-container-p2');

            // Render Page 1 Quiz (Q1-Q3)
            QuizEngine.getQuestionsPage1().forEach(q => {
                containerP1.appendChild(createQuestionElement(q, 1));
            });

            // Render Page 2 Quiz (Q4-Q7)
            QuizEngine.getQuestionsPage2().forEach(q => {
                containerP2.appendChild(createQuestionElement(q, 2));
            });
        }

        function createQuestionElement(qData, pageIdx) {
            const block = document.createElement('div');
            block.className = 'question-block';
            
            const qText = document.createElement('div');
            qText.className = 'question-text';
            qText.innerText = qData.q;
            block.appendChild(qText);

            const optionsList = document.createElement('div');
            optionsList.className = 'options-list';

            qData.options.forEach((opt, idx) => {
                const optItem = document.createElement('div');
                optItem.className = 'option-item';
                optItem.innerText = opt.text;
                
                // Event saat opsi dipilih
                optItem.onclick = function() {
                    // Bersihkan seleksi pada pertanyaan ini
                    optionsList.querySelectorAll('.option-item').forEach(el => el.classList.remove('selected'));
                    // Tambah kelas selected pada item terpilih
                    optItem.classList.add('selected');
                    
                    // Simpan ke state engine privat
                    QuizEngine.setAnswer(qData.id, idx);
                    
                    // Cek validasi tombol lanjut hlaman
                    validateNextButton(pageIdx);
                };
                optionsList.appendChild(optItem);
            });

            block.appendChild(optionsList);
            return block;
        }

        function validateNextButton(pageIdx) {
            const isComplete = QuizEngine.checkPageComplete(pageIdx);
            if(pageIdx === 1) {
                document.getElementById('btn-next-p1').disabled = !isComplete;
            } else if(pageIdx === 2) {
                document.getElementById('btn-next-p2').disabled = !isComplete;
            }
        }

        // Hitung Skor Akhir & Tampilkan Hasil
        function calculateAndShowResult() {
            const finalScore = QuizEngine.calculateScore();
            const resultData = QuizEngine.getGradeData(finalScore);

            // Set data visual ke halaman grade
            document.getElementById('display-score').innerText = finalScore;
            document.getElementById('display-grade').innerText = resultData.grade;
            document.getElementById('display-message').innerText = resultData.msg;

            // Alihkan ke Page Hasil akhir
            navigateTo('screen-5');
        }

        // Inisialisasi awal saat aplikasi dibuka
        window.onload = function() {
            renderQuiz();
